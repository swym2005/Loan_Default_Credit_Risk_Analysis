def classify_credit_risk(description):
    if 'Very Low Risk' in description:
        return 'Very Low Risk'
    elif 'Low Risk' in description:
        return 'Low Risk'
    elif 'Medium Risk' in description:
        return 'Medium Risk'
    elif 'Very High Risk' in description:
        return 'Very High Risk'
    elif 'High Risk' in description:
        return 'High Risk'
    elif description == 'No Bureau History Available':
        return 'No Bureau History'
    elif description.startswith('Not Scored'):
        return 'Not Scored'
    else:
        return 'Other'

# In general function to generate the analysis
def default_summary(data, column):
    result = (
        data.groupby(column, observed=True, dropna=False)
        .agg(
            Total_Loans=('loan_default', 'count'),
            Defaulted_Loans=('loan_default', 'sum'),
            Default_Rate=('loan_default', 'mean')
        )
    )

    result['Default_Rate'] *= 100

    return result.round(2)

    df['disbursed_amount'],
    q=4,
    labels=['Q1 - Smallest', 'Q2', 'Q3', 'Q4 - Largest']
)

# Calculate default rates for each group
loan_size_analysis = (
    df.groupby('Loan_Size_Group', observed=True)
    .agg(
        Total_Loans=('loan_default', 'count'),
        Defaulted_Loans=('loan_default', 'sum'),
        Average_Loan=('disbursed_amount', 'mean'),
        Default_Rate=('loan_default', 'mean')
    )
)
